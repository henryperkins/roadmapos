# RoadmapOS Governance Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Establish the typed, tamper-evident governance core and pilot registry that every later GitHub, agent, audit, and reporting workflow depends on.

**Architecture:** Build a pnpm TypeScript monorepo. Keep domain validation and pure reducers in `packages/domain`, policy parsing and enforcement in `packages/policy`, persistence in `packages/db`, fixtures in `packages/testkit`, and HTTP/job entry points in `apps/control-plane`. PostgreSQL is the only required stateful service; a transactional job table and append-only hash chain avoid a second queue or event-store dependency.

**Tech Stack:** TypeScript, Node.js, pnpm, Fastify, PostgreSQL, Drizzle ORM, Zod, YAML, Vitest, Docker Compose.

## Global Constraints

- Follow every global constraint in `docs/superpowers/plans/2026-07-20-roadmapos-governed-orchestration-pilot-master-REVIEW-REVISION.md`.
- Store all timestamps as UTC `timestamptz` and serialize them as ISO 8601 strings.
- Store GitHub numeric identifiers as decimal strings at TypeScript boundaries and `bigint` in PostgreSQL.
- Use UUID v4 identifiers generated by the application.
- Canonical hashes use SHA-256 over UTF-8 canonical JSON with recursively sorted object keys and unchanged array order.
- Hidden classifications and expected challenge labels use AES-256-GCM with a 32-byte key supplied through `ARTIFACT_ENCRYPTION_KEY_BASE64`.
- The append-only audit chain is the integrity record; mutable projection tables are convenience views and never replace audit events.
- No table containing approvals, frozen packages, classifications, incidents, or audit events permits `UPDATE` or `DELETE` through the application role.

## File structure

- `packages/domain/`: Zod schemas, canonical serialization, hashes, encrypted-artifact types, signed-link types, and pure shared domain values. It never imports infrastructure packages.
- `packages/policy/`: `.roadmapos/policy.yml` parsing, GitHub-protection composition, and pure action evaluation.
- `packages/db/`: PostgreSQL migrations, Drizzle schema, transaction helpers, append-only audit chaining, and durable jobs.
- `packages/testkit/`: deterministic database reset and fixtures shared by integration tests; it contains no production decision logic.
- `apps/control-plane/`: Fastify configuration, internal role authentication, pilot registry routes, and process entry points.

```ts
// packages/domain/src/canonical-json.ts
export type JsonPrimitive = string | number | boolean | null;
export type JsonValue = JsonPrimitive | JsonValue[] | { [key: string]: JsonValue };

function normalize(value: unknown): JsonValue {
  if (value === null || typeof value === "string" || typeof value === "boolean") {
    return value;
  }
  if (typeof value === "number" && Number.isFinite(value)) {
    return value;
  }
  if (Array.isArray(value)) {
    return value.map(normalize);
  }
  if (typeof value === "object") {
    const entries = Object.entries(value as Record<string, unknown>)
      .sort(([left], [right]) => left.localeCompare(right))
      .map(([key, entry]) => [key, normalize(entry)] as const);
    return Object.fromEntries(entries);
  }
  throw new TypeError(`Unsupported JSON value: ${String(value)}`);
}

export function canonicalJson(value: unknown): string {
  return JSON.stringify(normalize(value));
}
```

```ts
// packages/domain/src/hash.ts
import { createHash } from "node:crypto";
import { canonicalJson } from "./canonical-json.js";

export function sha256Text(value: string): string {
  return createHash("sha256").update(value, "utf8").digest("hex");
}

export function sha256Canonical(value: unknown): string {
  return sha256Text(canonicalJson(value));
}
```

    summary: z.string().min(1)

  z.object({
    kind: z.literal("open_draft_pr"),

    summary: z.string().min(1),

// packages/domain/src/secret-box.ts
import { createCipheriv, createDecipheriv, randomBytes } from "node:crypto";
import { canonicalJson } from "./canonical-json.js";

export interface SealedArtifact {
  version: 1;
  iv: string;
  tag: string;
  ciphertext: string;
}

export class SecretBox {
  readonly #key: Buffer;

  constructor(base64Key: string) {
    this.#key = Buffer.from(base64Key, "base64");
    if (this.#key.length !== 32) {
      throw new Error("ARTIFACT_ENCRYPTION_KEY_BASE64 must decode to 32 bytes");
    }
  }

  seal(value: unknown, associatedData: string): SealedArtifact {
    const iv = randomBytes(12);
    const cipher = createCipheriv("aes-256-gcm", this.#key, iv);
    cipher.setAAD(Buffer.from(associatedData, "utf8"));
    const ciphertext = Buffer.concat([
      cipher.update(canonicalJson(value), "utf8"),
      cipher.final()
    ]);
    return {
      version: 1,
      iv: iv.toString("base64"),
      tag: cipher.getAuthTag().toString("base64"),
      ciphertext: ciphertext.toString("base64")
    };
  }

  open<T>(artifact: SealedArtifact, associatedData: string): T {
    try {
      const decipher = createDecipheriv("aes-256-gcm", this.#key, Buffer.from(artifact.iv, "base64"));
      decipher.setAAD(Buffer.from(associatedData, "utf8"));
      decipher.setAuthTag(Buffer.from(artifact.tag, "base64"));
      const plaintext = Buffer.concat([
        decipher.update(Buffer.from(artifact.ciphertext, "base64")),
        decipher.final()
      ]).toString("utf8");
      return JSON.parse(plaintext) as T;
    } catch {
      throw new Error("Unable to decrypt artifact");
    }
  }
}

// packages/domain/src/signed-link.ts
import { createHmac, timingSafeEqual } from "node:crypto";
import { z } from "zod";
import { canonicalJson } from "./canonical-json.js";

const PilotLinkPayloadSchema = z.object({
  subject: z.enum(["effort_confirmation", "monitoring_confirmation", "trust_survey", "review_ready", "quality_assessment"]),
  subjectId: z.string().min(1),
  githubUserId: z.string().regex(/^\d+$/),
  expiresAt: z.number().int().positive()
});
export type PilotLinkPayload = z.infer<typeof PilotLinkPayloadSchema>;

function encode(value: string): string {
  return Buffer.from(value, "utf8").toString("base64url");
}

function signature(body: string, secret: string): Buffer {
  return createHmac("sha256", secret).update(body, "utf8").digest();
}

export function signPilotLink(payload: PilotLinkPayload, secret: string): string {
  const body = encode(canonicalJson(PilotLinkPayloadSchema.parse(payload)));
  return `${body}.${signature(body, secret).toString("base64url")}`;
}

export function verifyPilotLink(token: string, secret: string, now = Date.now()): PilotLinkPayload {
  const [body, provided] = token.split(".");
  if (!body || !provided) throw new Error("Malformed signed link");
  const expected = signature(body, secret);
  const actual = Buffer.from(provided, "base64url");
  if (actual.length !== expected.length || !timingSafeEqual(actual, expected)) {
    throw new Error("Invalid signed link");
  }
  const payload = PilotLinkPayloadSchema.parse(JSON.parse(Buffer.from(body, "base64url").toString("utf8")));
  if (payload.expiresAt < now) throw new Error("Link has expired");
  return payload;
}

export interface AuditEventInput {
  pilotId: string;
  eventType: string;
  actorType: "system" | "agent" | "maintainer" | "operator" | "auditor" | "github";
  actorId: string;
  payload: unknown;
}

export interface AuditEventRecord {
  id: string;
  sequence: number;
  previousHash: string | null;
  eventHash: string;
  createdAt: string;
}

export type SqlTransaction = Parameters<Parameters<Database["sql"]["begin"]>[0]>[0];

export async function appendAuditEventTx(transaction: SqlTransaction, input: AuditEventInput): Promise<AuditEventRecord> {
  await transaction`SELECT pg_advisory_xact_lock(hashtext(${input.pilotId}))`;
  const [previous] = await transaction<{ sequence: number; event_hash: string }[]>`
    SELECT sequence, event_hash
    FROM pilot_events
    WHERE pilot_id = ${input.pilotId}::uuid
    ORDER BY sequence DESC
    LIMIT 1
  `;
  const sequence = (previous?.sequence ?? 0) + 1;
  const id = crypto.randomUUID();
  const createdAt = new Date().toISOString();
  const previousHash = previous?.event_hash ?? null;
  const eventHash = sha256Canonical({
    id,
    pilotId: input.pilotId,
    sequence,
    eventType: input.eventType,
    actorType: input.actorType,
    actorId: input.actorId,
    payload: input.payload,
    previousHash,
    createdAt
  });
  await transaction`
    INSERT INTO pilot_events
      (id, pilot_id, sequence, event_type, actor_type, actor_id, payload, previous_hash, event_hash, created_at)
    VALUES
      (${id}::uuid, ${input.pilotId}::uuid, ${sequence}, ${input.eventType}, ${input.actorType},
       ${input.actorId}, ${transaction.json(input.payload)}, ${previousHash}, ${eventHash}, ${createdAt}::timestamptz)
  `;
  return { id, sequence, previousHash, eventHash, createdAt };
}

    VALUES (${id}::uuid, ${jobType}, ${database.sql.json(payload)})
  `;
  return id;
}

export async function claimNextJob(database: Database, workerId: string, allowedTypes: string[]) {
  if (allowedTypes.length === 0) return null;
  return database.sql.begin(async (transaction) => {
    const [job] = await transaction<{
      id: string;
      job_type: string;
      payload: unknown;
      attempts: number;
      max_attempts: number;
    }[]>`
      SELECT id, job_type, payload, attempts, max_attempts
      FROM jobs
      WHERE status = 'queued'
        AND available_at <= now()
        AND job_type = ANY(${allowedTypes})
      ORDER BY created_at
      FOR UPDATE SKIP LOCKED
      LIMIT 1
    `;
    if (!job) return null;
    await transaction`
      UPDATE jobs
      SET status = 'running', locked_by = ${workerId}, locked_at = now(), attempts = attempts + 1
      WHERE id = ${job.id}::uuid
    `;
    return { id: job.id, jobType: job.job_type, payload: job.payload, attempt: job.attempts + 1, maxAttempts: job.max_attempts };
  });
}

export async function completeJob(database: Database, jobId: string): Promise<void> {
  await database.sql`UPDATE jobs SET status = 'complete', completed_at = now() WHERE id = ${jobId}::uuid`;
}

export async function failJob(database: Database, jobId: string, message: string, retryAt: Date | null): Promise<void> {
  await database.sql`
    UPDATE jobs
    SET status = CASE WHEN ${retryAt?.toISOString() ?? null}::timestamptz IS NULL THEN 'failed' ELSE 'queued' END,
        available_at = COALESCE(${retryAt?.toISOString() ?? null}::timestamptz, available_at),
        locked_by = NULL,
        locked_at = NULL,
        last_error = ${message}
    WHERE id = ${jobId}::uuid
  `;
}

import { z } from "zod";
import YAML from "yaml";

const RawPolicySchema = z.object({
  version: z.literal(1),
  authorization: z.object({
    minimum_approval_role: z.enum(["write", "maintain", "admin"]),
    execution_plan_quorum: z.number().int().min(1),
    escalation_quorum: z.number().int().min(1)
  }),
  protected_zones: z.array(z.object({ path: z.string().min(1), reason: z.string().min(1) })).default([]),
  dependencies: z.object({ additions_require_escalation: z.boolean().default(true) }).default({}),
  public_interfaces: z.object({ changes_require_escalation: z.boolean().default(true) }).default({}),
  execution: z.object({
    readable_paths: z.array(z.string().min(1)).min(1),
    validation_commands: z.array(z.object({
      id: z.string().regex(/^[a-z0-9][a-z0-9_-]*$/),
      argv: z.array(z.string().min(1)).min(1),
      timeout_seconds: z.number().int().min(1).max(3600)
    })).min(1)
  })
});

export interface RepositoryPolicy {
  version: 1;
  authorization: {
    minimumApprovalRole: "write" | "maintain" | "admin";
    executionPlanQuorum: number;
    escalationQuorum: number;
  };
  protectedZones: Array<{ path: string; reason: string }>;
  dependencies: { additionsRequireEscalation: boolean };
  publicInterfaces: { changesRequireEscalation: boolean };
  execution: {
    readablePaths: string[];
    validationCommands: Array<{ id: string; argv: string[]; timeoutSeconds: number }>;
  };
}

// Portal-generated policy templates supply maintain/1/1, but committed policy files must state all three keys explicitly.
export function parseRepositoryPolicy(source: string): RepositoryPolicy {
  const raw = RawPolicySchema.parse(YAML.parse(source));
  return {
    version: 1,
    authorization: {
      minimumApprovalRole: raw.authorization.minimum_approval_role,
      executionPlanQuorum: raw.authorization.execution_plan_quorum,
      escalationQuorum: raw.authorization.escalation_quorum
    },
    protectedZones: raw.protected_zones,
    dependencies: { additionsRequireEscalation: raw.dependencies.additions_require_escalation },
    publicInterfaces: { changesRequireEscalation: raw.public_interfaces.changes_require_escalation },
    execution: {
      readablePaths: raw.execution.readable_paths,
      validationCommands: raw.execution.validation_commands.map((command) => ({
        id: command.id,
        argv: command.argv,
        timeoutSeconds: command.timeout_seconds
      }))
    }
  };
}

// packages/policy/src/evaluate-action.ts
import { minimatch } from "minimatch";
import type { ActionProposal } from "@roadmapos/domain";
import type { EffectivePolicy } from "./effective-policy.js";
import { PLATFORM_HARD_STOP_ACTIONS } from "./platform-hard-stops.js";

export interface ActionPolicyEvaluation {
  boundary: "none" | "repository_protected" | "platform_hard_stop";
  requiresEscalation: boolean;
  matchedRules: string[];
}

export function evaluateAction(policy: EffectivePolicy, action: ActionProposal): ActionPolicyEvaluation {
  const actionKind = (action as { kind: string }).kind;
  if (PLATFORM_HARD_STOP_ACTIONS.has(actionKind) || ![
    "read_files", "apply_patch", "run_validation", "commit_and_push", "open_draft_pr", "finish"
  ].includes(actionKind)) {
    return { boundary: "platform_hard_stop", requiresEscalation: false, matchedRules: [`platform:${actionKind}`] };
  }

  const matchedRules: string[] = [];
  if (action.kind === "apply_patch") {
    for (const path of action.affectedPaths) {
      for (const zone of policy.repository.protectedZones) {
        if (minimatch(path, zone.path, { dot: true })) matchedRules.push(`protected_zone:${zone.path}`);
      }
    }
    if (action.changesDependencies && policy.repository.dependencies.additionsRequireEscalation) {
      matchedRules.push("dependencies:additions_require_escalation");
    }
    if (action.changesPublicInterface && policy.repository.publicInterfaces.changesRequireEscalation) {
      matchedRules.push("public_interfaces:changes_require_escalation");
    }
    if (action.changesSecurityBehavior) matchedRules.push("security_behavior:agent_declared_change");
  }

  return matchedRules.length > 0
    ? { boundary: "repository_protected", requiresEscalation: true, matchedRules }
    : { boundary: "none", requiresEscalation: false, matchedRules: [] };
}

// apps/control-plane/src/auth.ts
import { createHash, timingSafeEqual } from "node:crypto";
import type { FastifyInstance, FastifyRequest } from "fastify";
import type { ControlPlaneConfig } from "./config.js";

export type InternalRole = "operator" | "auditor";

declare module "fastify" {
  interface FastifyRequest { internalRole?: InternalRole }
}

function matches(token: string, expectedHex: string): boolean {
  const actual = createHash("sha256").update(token, "utf8").digest();
  const expected = Buffer.from(expectedHex, "hex");
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}

export async function registerInternalAuth(app: FastifyInstance, config: ControlPlaneConfig): Promise<void> {
  app.decorateRequest("internalRole");
  app.addHook("preHandler", async (request, reply) => {
    if (!request.url.startsWith("/internal/")) return;
    const value = request.headers.authorization;
    const token = value?.startsWith("Bearer ") ? value.slice(7) : "";
    if (matches(token, config.operatorTokenSha256)) request.internalRole = "operator";
    else if (matches(token, config.auditorTokenSha256)) request.internalRole = "auditor";
    else return reply.code(401).send({ error: "unauthorized" });
  });
}

export function requireRole(request: FastifyRequest, role: InternalRole): void {
  if (request.internalRole !== role) {
    const error = new Error("forbidden") as Error & { statusCode: number };
    error.statusCode = 403;
    throw error;
  }
}

import { sha256Canonical } from "@roadmapos/domain";
import { parseRepositoryPolicy } from "@roadmapos/policy";

export class PilotRegistry {
  constructor(private readonly database: Database) {}

  async createPilot(input: string | { id?: string; name: string }): Promise<string> {
    const name = typeof input === "string" ? input : input.name;
    const id = typeof input === "string" ? crypto.randomUUID() : input.id ?? crypto.randomUUID();
    await this.database.db.insert(pilotStudies).values({ id, name, status: "draft" });
    await appendAuditEvent(this.database, { pilotId: id, eventType: "pilot.created", actorType: "operator", actorId: "operator", payload: { name } });
    return id;
  }

  async registerRepository(input: {
    id?: string; pilotId: string; githubRepositoryId: string; installationId: string; owner: string; name: string; defaultBranch: string;
  }): Promise<string> {
    const id = input.id ?? crypto.randomUUID();
    await this.database.db.insert(repositories).values({
      id,
      pilotId: input.pilotId,
      githubRepositoryId: BigInt(input.githubRepositoryId),
      installationId: BigInt(input.installationId),
      owner: input.owner,
      name: input.name,
      defaultBranch: input.defaultBranch
    });
    await appendAuditEvent(this.database, { pilotId: input.pilotId, eventType: "repository.registered", actorType: "operator", actorId: "operator", payload: { repositoryId: id, owner: input.owner, name: input.name } });
    return id;

  async snapshotPolicy(input: { pilotId: string; repositoryId: string; sourceCommitSha: string; source: string; actorId: string }) {
    const policy = parseRepositoryPolicy(input.source);
    const policyHash = sha256Canonical(policy);
    const id = crypto.randomUUID();
    await this.database.db.insert(repositoryPolicyVersions).values({ id, repositoryId: input.repositoryId, sourceCommitSha: input.sourceCommitSha, policyJson: policy, policyHash, createdBy: input.actorId });
    await appendAuditEvent(this.database, { pilotId: input.pilotId, eventType: "policy.snapshotted", actorType: "operator", actorId: input.actorId, payload: { repositoryId: input.repositoryId, policyVersionId: id, policyHash } });
    return { id, policyHash, policy };
  }

  async createMatchedPair(input: {
    pairId?: string;
    itemAId?: string;
    itemBId?: string;
    pilotId: string;
    repositoryId: string;
    governanceBearing: boolean;
    matchingRecord: unknown;
    itemA: { issueNumber: number; nodeId: string; title: string; acceptanceCriteria: string[]; effortBand: string; governanceExposure: string };
    itemB: { issueNumber: number; nodeId: string; title: string; acceptanceCriteria: string[]; effortBand: string; governanceExposure: string };
  }): Promise<string> {
    const itemAId = input.itemAId ?? crypto.randomUUID();
    const itemBId = input.itemBId ?? crypto.randomUUID();
    const pairId = input.pairId ?? crypto.randomUUID();
    await this.database.db.transaction(async (transaction) => {
      await transaction.insert(workItems).values([
        { id: itemAId, repositoryId: input.repositoryId, githubIssueNumber: input.itemA.issueNumber, githubIssueNodeId: input.itemA.nodeId, title: input.itemA.title, acceptanceCriteria: input.itemA.acceptanceCriteria, expectedEffortBand: input.itemA.effortBand, expectedGovernanceExposure: input.itemA.governanceExposure },
        { id: itemBId, repositoryId: input.repositoryId, githubIssueNumber: input.itemB.issueNumber, githubIssueNodeId: input.itemB.nodeId, title: input.itemB.title, acceptanceCriteria: input.itemB.acceptanceCriteria, expectedEffortBand: input.itemB.effortBand, expectedGovernanceExposure: input.itemB.governanceExposure }
      ]);
      await transaction.insert(matchedPairs).values({ id: pairId, pilotId: input.pilotId, repositoryId: input.repositoryId, workItemAId: itemAId, workItemBId: itemBId, governanceBearing: input.governanceBearing, matchingRecord: input.matchingRecord });
    });
    return pairId;
  }

  async assignPair(
    pairId: string,
    conditionDraw: () => number = () => randomInt(2),
    orderTieBreakDraw: () => number = () => randomInt(2)
  ) {
    const result = await this.database.db.transaction(async (transaction) => {
      const [pair] = await transaction.select().from(matchedPairs).where(eq(matchedPairs.id, pairId)).for("update");
      if (!pair) throw new Error("Pair not found");
      if (pair.assignedAt) throw new Error("Pair is already assigned");

      const prior = await transaction.select({ assignmentRecord: matchedPairs.assignmentRecord })
        .from(matchedPairs)
        .where(eq(matchedPairs.pilotId, pair.pilotId));
      const firstCounts = prior.reduce((counts, row) => {
        const record = row.assignmentRecord as { firstCondition?: "governed_orchestration" | "interactive_copilot" } | null;
        if (record?.firstCondition) counts[record.firstCondition] += 1;
        return counts;
      }, { governed_orchestration: 0, interactive_copilot: 0 });

      const orderTieBreakValue = firstCounts.governed_orchestration === firstCounts.interactive_copilot
        ? orderTieBreakDraw()
        : null;
      if (orderTieBreakValue !== null && orderTieBreakValue !== 0 && orderTieBreakValue !== 1) {
        throw new Error("Order tie-break draw must be 0 or 1");
      }
      const firstCondition = firstCounts.governed_orchestration < firstCounts.interactive_copilot
        ? "governed_orchestration"
        : firstCounts.interactive_copilot < firstCounts.governed_orchestration
          ? "interactive_copilot"
          : orderTieBreakValue === 0 ? "governed_orchestration" : "interactive_copilot";

      const conditionDrawValue = conditionDraw();
      if (conditionDrawValue !== 0 && conditionDrawValue !== 1) throw new Error("Condition draw must be 0 or 1");
      const itemACondition = conditionDrawValue === 0 ? "governed_orchestration" : "interactive_copilot";
      const itemBCondition = itemACondition === "governed_orchestration" ? "interactive_copilot" : "governed_orchestration";
      const record = {
        conditionDraw: conditionDrawValue,
        orderTieBreakDraw: orderTieBreakValue,
        orderAllocation: firstCounts.governed_orchestration === firstCounts.interactive_copilot ? "random_tie_break" : "blocked_balance",
        itemACondition,
        itemBCondition,
        firstCondition,
        itemAOrder: itemACondition === firstCondition ? 1 : 2,
        itemBOrder: itemBCondition === firstCondition ? 1 : 2,
        assignedAt: new Date().toISOString()
      } as const;
      const assignmentHash = sha256Canonical(record);
      await transaction.update(workItems).set({ condition: record.itemACondition, executionOrder: record.itemAOrder }).where(eq(workItems.id, pair.workItemAId));
      await transaction.update(workItems).set({ condition: record.itemBCondition, executionOrder: record.itemBOrder }).where(eq(workItems.id, pair.workItemBId));
      await transaction.update(matchedPairs).set({ assignmentRecord: record, assignmentHash, assignedAt: new Date(record.assignedAt) }).where(eq(matchedPairs.id, pairId));
      return { pilotId: pair.pilotId, ...record, assignmentHash };
    });
    await appendAuditEvent(this.database, { pilotId: result.pilotId, eventType: "pair.assigned", actorType: "system", actorId: "assignment-service", payload: { pairId, assignmentHash: result.assignmentHash, firstCondition: result.firstCondition } });
    const { pilotId: _pilotId, ...publicResult } = result;
    return publicResult;
  }
